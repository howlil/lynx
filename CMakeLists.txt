cmake_minimum_required(VERSION 3.20)
project(lynx_m0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

add_executable(lynx_core_tests tests/core_tests.cpp)
target_include_directories(lynx_core_tests PRIVATE src/core)
add_test(NAME core_contract COMMAND lynx_core_tests)

if(WIN32)
  add_executable(lynx_m0 WIN32 src/win32/main.cpp)
  target_compile_definitions(lynx_m0 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(lynx_m0 PRIVATE gdiplus user32 gdi32 shell32 uiautomationcore oleaut32 advapi32 windowsapp)
  if(MSVC)
    target_compile_options(lynx_m0 PRIVATE /W4 /permissive- /EHsc)
    target_compile_options(lynx_core_tests PRIVATE /W4 /permissive- /EHsc)
  endif()
endif()
